RP2040PIOUART

先理解四层职责

这套代码没有把所有事情塞进一个文件。PIO 汇编负责精确时序，资源管理器避免冲突，功能层把底层细节包装成 C 接口，App 最终只负责选择配置和调用接口。

App决定引脚、波特率和使用场景
fun_pio_uart申请资源、配置状态机、提供收发 API
pio_manager统计 SM 和 32 条指令空间
uart.pio按固定周期输出或采样电平
PIO 硬件独立于 CPU 执行状态机程序
已经验证：PIO0 上同时加载 TX 与 RX 程序、申请两个状态机，GP6 发送到 GP7 后逐字节校验通过。

  3. hardware_pio 提供 claim、加载程序和状态机控制函数。

资源表的数据模型

#define PIO_COUNT 2
#define PIO_SM_COUNT 4
#define PIO_PROGRAM_MAX_COUNT 8

#define PIO_MANAGER_ID_TO_BLOCK(id) \
    ((uint8_t)((id) / PIO_SM_COUNT))

#define PIO_MANAGER_ID_TO_SM(id) \
    ((uint8_t)((id) % PIO_SM_COUNT))

RP2040 有 PIO0、PIO1 两个块，每块 4 个状态机。管理器把它们编码为全局编号：

全局 IDPIO 块状态机
0～3PIO0SM0～SM3
4～7PIO1SM0～SM3
 
typedef struct {
    bool used;          // 软件是否认为该 SM 已占用
    const char *label;  // 谁在使用，例如 "pio_uart_tx"
} pio_sm_state_t;

typedef struct {
    bool loaded;
    const char *name;
    uint8_t offset;
    uint8_t length;
    uint8_t ref_count;
    const struct pio_program *program;
} pio_program_state_t;

typedef struct {
    PIO pio;
    pio_sm_state_t sm[PIO_SM_COUNT];
    pio_program_state_t programs[PIO_PROGRAM_MAX_COUNT];
    uint8_t program_count;
    uint8_t instruction_used;
} pio_state_t;

typedef struct {
    pio_state_t block[PIO_COUNT];
} pio_manager_t;
状态机和程序不是一回事：状态机是运行者；程序位于每个 PIO 独有的 32 条指令 RAM 中。同一个 TX 程序可以被多个 SM 共用，所以需要 ref_count。
pio/pio_manager.c

状态机和程序的生命周期

1. 初始化软件资源表

void pio_manager_init(pio_manager_t *manager)
{
    if (manager == NULL)
        return;

    manager->block[0].pio = pio0;
    manager->block[1].pio = pio1;

    for (uint8_t pio_index = 0; pio_index < PIO_COUNT; ++pio_index) {
        for (uint8_t sm = 0; sm < PIO_SM_COUNT; ++sm) {
            manager->block[pio_index].sm[sm].used = false;
            manager->block[pio_index].sm[sm].label = NULL;
        }

        manager->block[pio_index].program_count = 0;
        manager->block[pio_index].instruction_used = 0;
        // programs[] 的每个记录也全部清零
    }
}

它初始化的是软件账本。系统应只初始化一次；运行途中重复初始化会丢失软件记录，但 SDK 中真实 claim 的状态不会自动消失。

2. 从指定 PIO 申请状态机

int pio_manager_claim_sm_from(pio_manager_t *manager,
                              uint8_t pio_index,
                              const char *owner_name)
{
    PIO pio = manager->block[pio_index].pio;

    for (uint8_t sm = 0; sm < PIO_SM_COUNT; ++sm) {
        pio_sm_state_t *state = manager->block[pio_index].sm + sm;

        if (!state->used && !pio_sm_is_claimed(pio, sm)) {
            pio_sm_claim(pio, sm);
            state->used = true;
            state->label = owner_name;
            return pio_index * PIO_SM_COUNT + sm;
        }
    }
    return -1;
}

这里做双重检查：自己的 used 账本和 Pico SDK 的真实 claim 状态必须都空闲。

3. 加载程序并共享

int existing = pio_manager_find_program(
    manager, pio_index, program_name);

if (existing >= 0) {
    block->programs[existing].ref_count++;
    return block->programs[existing].offset;
}

if (!pio_can_add_program(block->pio, program))
    return -1;

uint offset = pio_add_program(block->pio, program);
state->offset = (uint8_t)offset;
state->length = (uint8_t)program->length;
state->ref_count = 1;

第二个 UART TX 状态机加载同名程序时不会再占指令空间，只增加引用计数并复用原来的 offset。

4. 释放最后一个引用

if (state->ref_count > 1) {
    state->ref_count--;
    return true;
}

pio_remove_program(block->pio,
                   state->program,
                   state->offset);
block->instruction_used -= state->length;

for (uint8_t i = (uint8_t)index;
     i + 1 < block->program_count;
     ++i) {
    block->programs[i] = block->programs[i + 1];
}
block->program_count--;

只有最后一个使用者关闭后才删除 PIO 指令。数组前移保证有效记录始终位于 0..program_count-1。

pio/uart.pio

真正执行 UART 时序的程序

TX：起始位、8 个数据位、停止位

.program uart_tx
.side_set 1 opt

.wrap_target
    ; TX FIFO 为空时阻塞，并让线路保持高电平。
    ; [7] 加上指令本身，共 8 个 PIO 时钟周期。
    pull block          side 1 [7]

    ; 起始位拉低。X=7 表示后面循环 8 次。
    set x, 7            side 0 [7]

bitloop:
    ; 从 OSR 最低位开始输出，指令本身占 1 周期。
    out pins, 1

    ; jmp 占 1 周期，[6] 再延迟 6 周期。
    ; out + jmp 总计 8 周期，也就是一个数据位。
    jmp x-- bitloop            [6]
.wrap

RX：等待下降沿并在位中心采样

.program uart_rx

.wrap_target
    ; UART 空闲为高，起始位为低。
    wait 0 pin 0

    ; 等待到第一个数据位中央，并准备接收 8 位。
    set x, 7 [10]

bitloop:
    ; 将当前 RX 电平移入 ISR。
    in pins, 1

    ; 每隔 8 个 PIO 周期采样下一位。
    jmp x-- bitloop [6]

    ; 8 位完成后写入 RX FIFO。
    push block
.wrap

fun/fun_pio_uart.c

从配置参数到运行中的状态机

TX open 的资源申请顺序

// 1. 校验 PIO、GPIO 和波特率
if (pio_index >= PIO_COUNT ||
    tx_gpio >= NUM_BANK0_GPIOS ||
    baudrate == 0) {
    return false;
}

// 2. 根据 8 个 PIO 周期/位计算分频
float clkdiv = (float)clock_get_hz(clk_sys) /
               ((float)baudrate * 8.0f);

// 3. 申请一个状态机
int sm_id = pio_manager_claim_sm_from(
    manager, pio_index, "pio_uart_tx");

// 4. 将 TX 程序加载进同一个 PIO
int offset = pio_manager_load_program(
    manager, pio_index, "uart_tx", &uart_tx_program);

// 5. 如果加载失败，必须归还刚申请的状态机
if (offset < 0) {
    pio_manager_release_sm(manager, pio_index, sm);
    return false;
}
申请多个资源时要有回滚路径。否则程序加载失败后，状态机会永远留在“已占用”状态。

配置 TX 状态机

pio_sm_config config =
    uart_tx_program_get_default_config((uint)offset);

sm_config_set_out_pins(&config, tx_gpio, 1);
sm_config_set_sideset_pins(&config, tx_gpio);
sm_config_set_out_shift(&config, true, false, 32);
sm_config_set_clkdiv(&config, clkdiv);

pio_gpio_init(pio, tx_gpio);
pio_sm_set_enabled(pio, sm, false);
pio_sm_clear_fifos(pio, sm);
pio_sm_init(pio, sm, (uint)offset, &config);

// UART 空闲必须为高，随后才启用状态机。
pio_sm_set_pins_with_mask(pio, sm,
                          1u << tx_gpio,
                          1u << tx_gpio);
pio_sm_set_pindirs_with_mask(pio, sm,
                             1u << tx_gpio,
                             1u << tx_gpio);
pio_sm_set_enabled(pio, sm, true);

shift_right=true 使 OSR 从最低位开始输出，符合 UART 的 LSB first。

配置 RX 状态机

pio_sm_config config =
    uart_rx_program_get_default_config((uint)offset);

sm_config_set_in_pins(&config, rx_gpio);
sm_config_set_in_shift(&config, true, false, 32);
sm_config_set_fifo_join(&config, PIO_FIFO_JOIN_RX);
sm_config_set_clkdiv(&config, clkdiv);

pio_gpio_init(pio, rx_gpio);
gpio_pull_up(rx_gpio);
pio_sm_set_consecutive_pindirs(pio, sm, rx_gpio, 1, false);
pio_sm_init(pio, sm, (uint)offset, &config);
pio_sm_set_enabled(pio, sm, true);

PIO_FIFO_JOIN_RX 将 TX/RX 两侧 FIFO 合并成更深的 RX FIFO。RX 不发送数据，所以可以把空间全部给接收端。

为什么读取后要右移 24 位

return (int)(pio_sm_get(rx->pio, rx->sm) >> 24);

RX 配置为向右移位，8 次采样最终位于 ISR 的最高 8 位；FIFO 读出的是 32 位，因此需要右移 24 位得到字节。

关闭时按相反顺序释放

pio_sm_set_enabled(pio, sm, false);
pio_sm_clear_fifos(pio, sm);
gpio_set_function(gpio, GPIO_FUNC_NULL);

pio_manager_release_sm(manager, pio_index, sm);
pio_manager_release_program(manager, pio_index, "uart_tx");
memset(channel, 0, sizeof(*channel));

先停止硬件，再释放软件资源。程序的引用计数归零后，管理器才会从指令 RAM 删除它。

在业务代码中的正确调用顺序

static pio_manager_t g_pio_manager;
static fun_pio_uart_tx_t g_tx;
static fun_pio_uart_rx_t g_rx;

void system_pio_init(void)
{
    // 整个系统只调用一次
    pio_manager_init(&g_pio_manager);
}

bool channel_open(void)
{
    if (!fun_pio_uart_rx_open(
            &g_rx, &g_pio_manager, 0, 7, 115200)) {
        return false;
    }

    if (!fun_pio_uart_tx_open(
            &g_tx, &g_pio_manager, 0, 6, 115200)) {
        fun_pio_uart_rx_close(&g_rx); // 回滚 RX
        return false;
    }

    return true;
}

void channel_poll(void)
{
    uint8_t buffer[64];
    size_t count = fun_pio_uart_rx_read(
        &g_rx, buffer, sizeof(buffer));

    if (count > 0) {
        // 处理收到的数据
    }
}

void channel_send(const uint8_t *data, size_t length)
{
    fun_pio_uart_tx_write(&g_tx, data, length);
}

void channel_close(void)
{
    fun_pio_uart_tx_close(&g_tx);
    fun_pio_uart_rx_close(&g_rx);
}
正式接入 App 时，管理器应归系统所有；App 只持有 TX/RX 通道或句柄。否则两个 App 各自维护一份资源表，会认为同一个 SM 都是空闲的。

当前实现边界

项目当前状态后续方向
数据格式固定 8-N-1另写 7 位、校验位版本或在 PIO 程序中扩展
TX 写入阻塞 FIFO 写入增加非阻塞接口或 DMA
RX 读取轮询 FIFO增加 DMA 环形缓冲
错误检测不检查停止位和校验扩展 PIO 程序并上报 framing/parity error
资源所有权接口接收 manager 指针系统创建唯一管理器并向各功能层传递
引脚冲突仅排除 GP0/GP1接入 config.h 的功能引脚分配表
